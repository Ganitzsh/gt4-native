cmake_minimum_required(VERSION 3.21)
project(gt4_native VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GT4_STRICT_WARNINGS "Enable additional compiler diagnostics" OFF)
option(GT4_ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(gt4_guest_abi INTERFACE)
if(GT4_STRICT_WARNINGS)
  if(MSVC)
    target_compile_options(gt4_guest_abi INTERFACE /W4)
  else()
    target_compile_options(gt4_guest_abi INTERFACE -Wall -Wextra -Wpedantic)
  endif()
endif()
if(GT4_ENABLE_SANITIZERS)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU" OR WIN32)
    message(FATAL_ERROR "Sanitizers require a GCC/Clang Unix toolchain")
  endif()
  target_compile_options(gt4_guest_abi INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(gt4_guest_abi INTERFACE -fsanitize=address,undefined)
endif()
target_include_directories(gt4_guest_abi INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_executable(gt4_native platform/pc/main.cpp)
target_link_libraries(gt4_native PRIVATE gt4_guest_abi)
add_executable(gt4_elf_inspect tools/elf_inspect.cpp)
target_link_libraries(gt4_elf_inspect PRIVATE gt4_guest_abi)
include(CTest)
if(BUILD_TESTING)
  add_executable(gt4_guest_abi_test tests/guest_abi.cpp)
  target_link_libraries(gt4_guest_abi_test PRIVATE gt4_guest_abi)
  add_executable(gt4_guest_memory_test tests/guest_memory.cpp)
  target_link_libraries(gt4_guest_memory_test PRIVATE gt4_guest_abi)
  add_test(NAME bootstrap COMMAND gt4_native --self-test)
  add_test(NAME guest_abi COMMAND gt4_guest_abi_test)
  add_test(NAME guest_memory COMMAND gt4_guest_memory_test)
  add_executable(gt4_guest_dispatch_test tests/guest_dispatch.cpp)
  target_link_libraries(gt4_guest_dispatch_test PRIVATE gt4_guest_abi)
  add_test(NAME guest_dispatch COMMAND gt4_guest_dispatch_test)
  add_executable(gt4_guest_string_test tests/guest_string.cpp)
  target_link_libraries(gt4_guest_string_test PRIVATE gt4_guest_abi)
  add_test(NAME guest_string COMMAND gt4_guest_string_test)
  add_executable(gt4_native_callbacks_test tests/native_callbacks.cpp)
  target_link_libraries(gt4_native_callbacks_test PRIVATE gt4_guest_abi)
  add_test(NAME native_callbacks COMMAND gt4_native_callbacks_test)
  add_executable(gt4_guest_address_space_test tests/guest_address_space.cpp)
  target_link_libraries(gt4_guest_address_space_test PRIVATE gt4_guest_abi)
  add_test(NAME guest_address_space COMMAND gt4_guest_address_space_test)
  add_executable(gt4_elf32_test tests/elf32.cpp)
  target_link_libraries(gt4_elf32_test PRIVATE gt4_guest_abi)
  add_test(NAME elf32 COMMAND gt4_elf32_test)
  add_executable(gt4_portable_primitives_test tests/portable_primitives.cpp)
  target_link_libraries(gt4_portable_primitives_test PRIVATE gt4_guest_abi)
  add_test(NAME portable_primitives COMMAND gt4_portable_primitives_test)
  add_executable(gt4_robustness_test tests/robustness.cpp)
  target_link_libraries(gt4_robustness_test PRIVATE gt4_guest_abi)
  add_test(NAME robustness COMMAND gt4_robustness_test)
  add_executable(gt4_elf_mutation_test tests/elf_mutation.cpp)
  target_link_libraries(gt4_elf_mutation_test PRIVATE gt4_guest_abi)
  add_test(NAME elf_mutation COMMAND gt4_elf_mutation_test)
  add_executable(gt4_upstream_mboolreader_test
    tests/upstream_mboolreader.cpp
    upstream/gt4/src/runtime/MboolReader/MboolReader__structor_19.cpp)
  target_link_libraries(gt4_upstream_mboolreader_test PRIVATE gt4_guest_abi)
  add_test(NAME upstream_mboolreader COMMAND gt4_upstream_mboolreader_test)

  # Tests use assert(); keep checks active even under Release builds.
  foreach(test_target IN ITEMS gt4_guest_abi_test gt4_guest_memory_test
      gt4_guest_dispatch_test gt4_guest_string_test gt4_native_callbacks_test
      gt4_guest_address_space_test gt4_elf32_test gt4_portable_primitives_test gt4_robustness_test gt4_elf_mutation_test gt4_upstream_mboolreader_test)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /UNDEBUG)
    else()
      target_compile_options(${test_target} PRIVATE -UNDEBUG)
    endif()
  endforeach()
endif()
